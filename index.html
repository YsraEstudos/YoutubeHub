    Firebase Firestore

    rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- RegBasicra 1: Perfis de Usuário ('users') ---
    // Cada usuário tem um documento na coleção 'users'
    // O ID do documento é o UID de autenticação do usuário
    match /users/{userId} {
      // Qualquer usuário logado pode ler o perfil de outro (para ver nome, foto, etc.)
      allow read: if request.auth != null;
      
      // Apenas o PRÓPRIO usuário pode criar ou atualizar seu perfil
      // (Verifica se o UID do logado é o mesmo do documento que ele quer mudar)
      allow write: if request.auth.uid == userId;
    }

    // --- Regra 2: Conteúdo Público (Ex: 'posts', 'produtos', 'artigos') ---
    // Uma coleção pública que todos podem ver
    match /posts/{postId} {
      
      // Qualquer pessoa (logada ou não) pode ler
      allow read: if true;
      
      // Apenas usuários logados podem CRIAR
      allow create: if request.auth != null;
      
      // Apenas o AUTOR original pode ATUALIZAR ou DELETAR
      // (IMPORTANTE: Veja a explicação abaixo!)
      allow update, delete: if request.auth.uid == resource.data.authorId;
    }

    // --- Regra 3: Dados Privados do Usuário (Ex: 'notas', 'carrinho') ---
    // Este padrão usa uma subcoleção dentro do documento do usuário
    // Ex: /users/{userId}/privateNotes/{noteId}
    match /users/{userId}/privateNotes/{noteId} {
      
      // Apenas o usuário "dono" (cujo ID está no caminho) pode ler ou escrever
      // Ninguém mais pode ver esses dados.
      allow read, write: if request.auth.uid == userId;
    }
  }
}

    Firebase Storage (Modo Produção)

    rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Regra para a coleção 'users'
    // Ela diz: "Qualquer usuário logado pode ler um perfil,
    // mas SÓ O PRÓPRIO DONO pode escrever (atualizar) seu perfil."

    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth.uid == userId;
    }

    // Lembre-se: Como você iniciou em "Modo de Produção",
    // todas as outras coleções estão bloqueadas por padrão.
    // Você terá que adicionar regras para 'posts', 'produtos', etc.,
    // da mesma forma que fizemos acima.
  }
}
